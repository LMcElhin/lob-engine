cmake_minimum_required(VERSION 3.20...4.2)
if(DEFINED SKBUILD_PROJECT_NAME)
    project(${SKBUILD_PROJECT_NAME} VERSION ${SKBUILD_PROJECT_VERSION} LANGUAGES CXX)
else()
    project(lob_engine VERSION 0.1.0 LANGUAGES CXX)
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)


add_library(lob_core STATIC cpp/src/order_book.cpp)
target_include_directories(lob_core PUBLIC cpp/include)
set_target_properties(lob_core PROPERTIES POSITION_INDEPENDENT_CODE ON)

if(MSVC)
    target_compile_options(lob_core PRIVATE /W4 /permissive-)
else()
    target_compile_options(lob_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

option(LOB_BUILD_PYTHON "Build the pybind11 Python extension" ON)
if(LOB_BUILD_PYTHON)
    set(PYBIND11_FINDPYTHON ON)
    find_package(pybind11 CONFIG REQUIRED)
    pybind11_add_module(_core cpp/src/bindings.cpp)
    target_link_libraries(_core PRIVATE lob_core)
    target_include_directories(_core PRIVATE cpp/include)
    install(TARGETS _core LIBRARY DESTINATION loblab RUNTIME DESTINATION loblab ARCHIVE DESTINATION loblab)
endif()

option(LOB_BUILD_BENCHMARK "Build native benchmark executable" OFF)
if(LOB_BUILD_BENCHMARK)
    add_executable(lob_benchmark cpp/benchmarks/benchmark.cpp)
    target_link_libraries(lob_benchmark PRIVATE lob_core)
    target_include_directories(lob_benchmark PRIVATE cpp/include)
endif()

option(LOB_BUILD_TESTS "Build native C++ tests" OFF)
if(LOB_BUILD_TESTS)
    enable_testing()
    add_executable(lob_cpp_tests cpp/tests/test_order_book.cpp)
    target_link_libraries(lob_cpp_tests PRIVATE lob_core)
    target_include_directories(lob_cpp_tests PRIVATE cpp/include)
    add_test(NAME lob_cpp_tests COMMAND lob_cpp_tests)
endif()
